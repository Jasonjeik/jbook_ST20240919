import pandas as pd
import matplotlib.pyplot as plt

def convert_volume(vol):
    if pd.isnull(vol): 
        return None
    if 'K' in vol:
        return float(vol.replace('K', '')) * 1000
    elif 'M' in vol:
        return float(vol.replace('M', '')) * 1000000
    else:
        try:
            return float(vol)
        except ValueError:
            return None

#Volatilidad
def volatility(w, at):
    vols = []
    for t in range(0, len(at) - w + 1):
        sqr = []
        r_list = at[t:t + w]
        miu = (1 / w) * (sum(r_list))
        
        for r in r_list:
            dif = r - miu
            sqr.append(dif ** 2)
        
        # Verificar si la varianza es negativa o NaN
        var = (1 / (w - 1)) * sum(sqr)
        if var < 0 or np.isnan(var) or np.isinf(var):
            vol = np.nan
        else:
            vol = np.sqrt(max(var, 0))
        
        vols.append(vol)
    
    vols = pd.Series(vols)
    return vols
